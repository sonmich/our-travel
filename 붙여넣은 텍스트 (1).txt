  const wrapper = document.getElementById('mapWrapper');
  const spots = document.querySelectorAll('.spot');
  let activeSpot = null;
  let isDragging = false;
  let startX = 0;
  let startY = 0;

  // 슬라이더 전역 변수
  let currentMediaList = [];
  let currentSlideIndex = 0;

  spots.forEach(spot => {
    spot.addEventListener('mousedown', startDrag);
    spot.addEventListener('touchstart', startDrag, { passive: false });
  });

  function startDrag(e) {
    activeSpot = this;
    isDragging = false;
    startX = e.clientX || e.touches[0].clientX;
    startY = e.clientY || e.touches[0].clientY;

    document.addEventListener('mousemove', drag);
    document.addEventListener('touchmove', drag, { passive: false });
    document.addEventListener('mouseup', stopDrag);
    document.addEventListener('touchend', stopDrag);
  }

  function drag(e) {
    if (!activeSpot) return;

    const clientX = e.clientX || e.touches[0].clientX;
    const clientY = e.clientY || e.touches[0].clientY;

    if (Math.abs(clientX - startX) > 5 || Math.abs(clientY - startY) > 5) {
      isDragging = true;
    }

    if (e.type === 'touchmove') e.preventDefault();

    const rect = wrapper.getBoundingClientRect();
    let left = ((clientX - rect.left) / rect.width) * 100;
    let top = ((clientY - rect.top) / rect.height) * 100;

    left = Math.max(0, Math.min(100, left));
    top = Math.max(0, Math.min(100, top));

    activeSpot.style.left = left.toFixed(1) + '%';
    activeSpot.style.top = top.toFixed(1) + '%';
  }

  function stopDrag() {
    if (activeSpot && !isDragging) {
      const name = activeSpot.getAttribute('data-name');
      const color = activeSpot.style.getPropertyValue('--spot-color');
      openModal(name, color);
    }

    activeSpot = null;
    document.removeEventListener('mousemove', drag);
    document.removeEventListener('touchmove', drag);
  }

  function openModal(name, color) {
    const data = spotData[name];
    if (!data) return;

    currentMediaList = data.media || [];
    currentSlideIndex = 0;
    updateSlideDisplay();

    document.getElementById('modalName').innerText = name;
    document.getElementById('modalDesc').innerText = data.desc;
    document.getElementById('modalDot').style.backgroundColor = color || '#2563eb';

    const tagsContainer = document.getElementById('modalTags');
    tagsContainer.innerHTML = '';
    data.tags.forEach(tag => {
      const span = document.createElement('span');
      span.className = 'modal-tag';
      span.innerText = tag;
      tagsContainer.appendChild(span);
    });

    document.getElementById('spotModal').classList.add('active');
  }

  /* 멀티미디어 슬라이드 화면 갱신 함수 */
  function updateSlideDisplay() {
    const imgEl = document.getElementById('modalImg');
    const videoEl = document.getElementById('modalVideo');
    const prevBtn = document.getElementById('prevBtn');
    const nextBtn = document.getElementById('nextBtn');
    const counter = document.getElementById('slideCounter');

    // 이전 슬라이드에 재생 중이던 동영상이 있다면 일시정지
    videoEl.pause();

    if (currentMediaList.length > 0) {
      const currentItem = currentMediaList[currentSlideIndex];

      if (currentItem.type === 'video') {
        imgEl.style.display = 'none';
        videoEl.style.display = 'block';
        videoEl.src = currentItem.url;
        
        const playPromise = videoEl.play();
        if (playPromise !== undefined) {
          playPromise.catch(() => {
            videoEl.muted = true;
            videoEl.play();
          });
        }
      } else {
        videoEl.style.display = 'none';
        imgEl.style.display = 'block';
        imgEl.src = currentItem.url;
      }
    }

    if (currentMediaList.length > 1) {
      prevBtn.style.display = 'flex';
      nextBtn.style.display = 'flex';
      counter.style.display = 'block';
      counter.innerText = `${currentSlideIndex + 1} / ${currentMediaList.length}`;
    } else {
      prevBtn.style.display = 'none';
      nextBtn.style.display = 'none';
      counter.style.display = 'none';
    }
  }

  /* 슬라이드 이동 처리 함수 */
  function changeSlide(direction) {
    if (currentMediaList.length <= 1) return;

    currentSlideIndex += direction;

    if (currentSlideIndex < 0) {
      currentSlideIndex = currentMediaList.length - 1;
    } else if (currentSlideIndex >= currentMediaList.length) {
      currentSlideIndex = 0;
    }

    updateSlideDisplay();
  }

  function openModalByName(name) {
    const spotEl = document.querySelector(`.spot[data-name="${name}"]`);
    const color = spotEl ? spotEl.style.getPropertyValue('--spot-color') : '#2563eb';
    openModal(name, color);
  }

  function closeModal() {
    const videoEl = document.getElementById('modalVideo');
    if (videoEl) {
      videoEl.pause();
    }
    document.getElementById('spotModal').classList.remove('active');
  }

  function closeModalOnOuterClick(e) {
    if (e.target.classList.contains('modal-overlay')) {
      closeModal();
    }
  }

  /* --- 전체 화면 미디어 확대 라이트박스 로직 --- */
  function openFullscreenMedia() {
    if (currentMediaList.length === 0) return;

    const currentItem = currentMediaList[currentSlideIndex];
    const fsModal = document.getElementById('fullscreenModal');
    const fsImg = document.getElementById('fullscreenImg');
    const fsVideo = document.getElementById('fullscreenVideo');

    if (currentItem.type === 'video') {
      fsImg.style.display = 'none';
      fsVideo.style.display = 'block';
      fsVideo.src = currentItem.url;
      fsVideo.play();
    } else {
      fsVideo.pause();
      fsVideo.style.display = 'none';
      fsImg.style.display = 'block';
      fsImg.src = currentItem.url;
    }

    fsModal.classList.add('active');
  }

  function closeFullscreenMedia() {
    const fsModal = document.getElementById('fullscreenModal');
    const fsVideo = document.getElementById('fullscreenVideo');

    if (fsVideo) {
      fsVideo.pause();
    }
    fsModal.classList.remove('active');
  }

  function closeFullscreenMediaOnOuterClick(e) {
    if (e.target.classList.contains('fullscreen-modal')) {
      closeFullscreenMedia();
    }
  }

  // 키보드 Esc 키 입력 시 확대 레이어/모달 닫기
  document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape') {
      closeFullscreenMedia();
      closeModal();
    }
  });

  function exportCoordinates() {
    let output = '';
    spots.forEach(spot => {
      const left = spot.style.left;
      const top = spot.style.top;
      const color = spot.style.getPropertyValue('--spot-color');
      const name = spot.getAttribute('data-name');
      
      let labelHtml = name === '구마모토' ? '\n  <div class="custom-label">구마모토</div>' : '';
      output += `<div class="spot" data-name="${name}" style="left: ${left}; top: ${top}; --spot-color: ${color};">\n  <div class="dot"></div><div class="pulse"></div>${labelHtml}\n</div>\n`;
    });
    
    navigator.clipboard.writeText(output).then(() => {
      alert('맞추신 마커 좌표가 클립보드에 복사되었습니다!');
    });
  }
